/* ========================================================================== Baby Boutique Ecuador - Catálogo de Productos desde productos.json ========================================================================== */ let listaProductos = []; async function obtenerProductos() { try { const respuesta = await fetch("data/productos.json"); if (respuesta.ok) { listaProductos = await respuesta.json(); renderizarProductos(listaProductos); } else { console.error("No se pudo cargar data/productos.json"); } } catch (error) { console.error("Error al leer la base de datos local:", error); } } function renderizarProductos(productosAMostrar = []) { const contenedor = document.getElementById('products-container') || document.getElementById('grid-productos'); if (!contenedor) return; contenedor.innerHTML = ''; if (productosAMostrar.length === 0) { contenedor.innerHTML = `

No se encontraron productos en esta categoría.

`; return; } productosAMostrar.forEach(prod => { const tarjeta = document.createElement('div'); tarjeta.className = 'product-card'; // Manejo flexible de imágenes (soporta array de imágenes o string directo) let imagenRuta = './img/placeholder.jpg'; if (Array.isArray(prod.imagenes) && prod.imagenes.length > 0) { imagenRuta = prod.imagenes[0]; } else if (prod.imagen) { imagenRuta = prod.imagen; } // Tallas por defecto si el producto no tiene especificadas const tallasEstandar = ["0 a 3 meses (Talla 0)", "3 a 6 meses (Talla 1)", "6 a 9 meses (Talla 2)", "9 a 12 meses (Talla 3)"]; const listaTallas = (prod.tallas && prod.tallas.length > 0) ? prod.tallas : tallasEstandar; tarjeta.innerHTML = `
CÓDIGO: ${prod.codigo || prod.id}

${prod.nombre}

 ${prod.colores && prod.colores.length > 0 ? ` ` : ''}
$${(prod.precio || 0).toFixed(2)} ${prod.precioMediaDocena ? `

Media Docena: $${prod.precioMediaDocena.toFixed(2)} c/u

` : ''}
`; contenedor.appendChild(tarjeta); }); } function prepararAgregar(idProducto) { const prod = listaProductos.find(p => p.id == idProducto || p.codigo == idProducto); if (!prod) return; const colorSelect = document.getElementById(`color-${idProducto}`); const tallaSelect = document.getElementById(`talla-${idProducto}`); const cantidadInput = document.getElementById(`cantidad-${idProducto}`); const cantidad = cantidadInput ? parseInt(cantidadInput.value) || 0 : 0; if (cantidad <= 0) { alert("Por favor selecciona al menos 1 prenda en la cantidad."); return; } let imagenRuta = prod.imagen || './img/placeholder.jpg'; if (Array.isArray(prod.imagenes) && prod.imagenes.length > 0) { imagenRuta = prod.imagenes[0]; } const productoAAgregar = { id: prod.id || prod.codigo, codigo: prod.codigo || prod.id, nombre: prod.nombre, precio: prod.precio, precioMediaDocena: prod.precioMediaDocena || prod.precio, precioDocena: prod.precioDocena || prod.precio, imagen: imagenRuta, color: colorSelect ? colorSelect.value : 'Único', talla: tallaSelect ? tallaSelect.value : 'Única', cantidad: cantidad }; if (typeof agregarAlCarrito === 'function') { agregarAlCarrito(productoAAgregar); } } function inicializarFiltros() { const botones = document.querySelectorAll('.cat-btn'); botones.forEach(btn => { btn.addEventListener('click', (e) => { botones.forEach(b => b.classList.remove('active')); e.currentTarget.classList.add('active'); const categoriaFiltro = e.currentTarget.getAttribute('data-category').trim().toLowerCase(); if (categoriaFiltro === 'todos') { renderizarProductos(listaProductos); } else { const filtrados = listaProductos.filter(p => { const catProd = (p.categoria || '').trim().toLowerCase(); return catProd === categoriaFiltro; }); renderizarProductos(filtrados); } }); }); } document.addEventListener('DOMContentLoaded', () => { obtenerProductos(); inicializarFiltros(); });
